# 认知副驾（Cognitive Copilot）

> 个人知识学习 AI 智能体 —— 通过持续对话与自主推理，把「收藏的碎片」转化为「可调用的认知」，并持续修正学习方向。
>
> 一句话价值：信息系统帮你「管住」已知；认知副驾帮你「照亮」未知，并「逼你」修正。

## 单一模型架构

| 模型服务 | 供应商 | 上下文窗口 | 关键能力 |
|------|--------|-----------|------|
| Anthropic Claude | Anthropic | 长上下文 | 深度推理 / 多轮对话 / 规划 / Tool Calls / 思考模式 / Prompt Caching |

## 五层能力梯度

| 层级 | 能力 | 解决的问题 |
|------|------|-----------|
| L1 反应层 | 模糊意图挖掘 | 让遗忘变成对话 |
| L2 关联层 | 隐性冲突检测 | 发现知识体系里的「内战」 |
| L3 生成层 | 苏格拉底式认知助产 | 生成你「该问但没问」的问题 |
| L4 规划层 | 动态路径自我修正 | 计划赶不上变化时自主重排 |
| L5 诊断层 | 归因式学习偏误诊断 | 解释行为背后的「病根」 |

"""

核心定位：通过持续对话与自主推理，帮助用户把"收藏的碎片"转化为"可调用的认知"，
         并持续修正学习方向。
一句话价值：信息系统帮你"管住"已知；认知副驾帮你"照亮"未知，并"逼你"修正。

from dataclasses import dataclass, field
from typing import List, Dict, Optional, Callable
from enum import Enum
from datetime import datetime, timedelta

# 一、核心数据结构
# =============================================================================

@dataclass
class KnowledgeItem:
    """知识条目——用户收藏/笔记的基本单位"""
    id: str
    title: str
    content: str
    source: str              # 来源：浏览器书签/Notion/Obsidian/飞书等
    url: Optional[str] = None
    tags: List[str] = field(default_factory=list)
    collected_at: datetime = field(default_factory=datetime.now)
    last_read_at: Optional[datetime] = None
    read_progress: float = 0.0   # 0-1，阅读进度


@dataclass
class LearningGoal:
    """学习目标——用户设定的阶段性目标"""
    description: str
    target_date: datetime
    priority: int = 1        # 1高 2中 3低
    achieved: bool = False


@dataclass
class LearningPlan:
    """学习计划——智能体生成的路径"""
    goal: LearningGoal
    tasks: List[Dict]        # [{week: 1, topic: "Pandas基础", status: "pending"}]
    created_at: datetime = field(default_factory=datetime.now)
    updated_at: datetime = field(default_factory=datetime.now)


@dataclass
class CognitiveDiagnosis:
    """认知诊断报告——偏误检测的输出"""
    pattern: str             # 行为模式：如"高收藏低完成"
    root_cause: str          # 归因诊断
    confidence: float        # 0-1
    suggested_action: str    # 可执行解决方案
    reasoning_chain: List[str]  # 推理链（CoT）


# =============================================================================
# 二、模型配置——单一 Claude 服务
# =============================================================================

@dataclass
class ModelEngine:
    """模型服务配置（单一供应商，保留后续可替换）"""
    name: str
    provider: str
    context_window: int      # Token
    max_output: int          # Token
    input_price: float       # 元/百万tokens（接入时回填实测值）
    output_price: float      # 元/百万tokens
    cache_hit_price: float   # 缓存命中输入价
    capabilities: List[str]  # 功能标签


# 单一模型服务：Anthropic Claude
CLAUDE = ModelEngine(
    name="claude",
    provider="Anthropic Claude",
    context_window=200_000,
    max_output=100_000,
    input_price=0.0,   # 占位：接入后以实测单价回填
    output_price=0.0,  # 占位：接入后以实测单价回填
    cache_hit_price=0.0,  # 占位：Prompt Caching 单价
    capabilities=["深度推理", "多轮对话", "复杂规划", "Tool Calls", "思考模式", "材料提取"]
)


class ModelGateway:
    """
    统一模型网关——单一 Claude 服务，按任务类型组织调用策略
    （所有任务统一走 CLAUDE，策略表仅描述任务语义，便于后续扩展）
    """
    
    # 任务类型 → 调用策略（同供应商内统一模型，此处仅登记任务语义）
    CALL_STRATEGIES = {
        "multi_turn_dialogue": "高性能 + 长上下文",   # 多轮对话
        "deep_reasoning": "思考模式",                 # 深度推理 / 冲突判断
        "conflict_detection": "思考模式",             # 冲突检测
        "plan_generation": "Tool Calls",              # 计划生成
        "causal_reasoning": "思考模式",               # 因果归因
        "cognitive_brief": "提示词模板",              # 认知简报 / 追问
        "batch_extraction": "批量 + Prompt Caching",  # 批量主张提取
        "topic_analysis": "批量 + Prompt Caching",    # 主题分析
    }
    
    @classmethod
    def route(cls, task_type: str) -> ModelEngine:
        """全部任务统一路由到 Claude"""
        return CLAUDE
    
    @classmethod
    def strategy(cls, task_type: str) -> str:
        """返回任务对应的调用策略描述"""
        return cls.CALL_STRATEGIES.get(task_type, "默认对话模型")
    
    @classmethod
    def estimate_cost(cls, task_type: str, input_tokens: int, output_tokens: int) -> float:
        """估算单次任务成本（基于 Claude 单价，接入后回填）"""
        engine = cls.route(task_type)
        cost = (input_tokens / 1_000_000) * engine.input_price + \
               (output_tokens / 1_000_000) * engine.output_price
        return cost

class AgentCapability(Enum):
    """AI智能体独特能力枚举——信息系统做不到的事"""
    SEMANTIC_REASONING = "语义推理"
    MULTI_TURN_CLARIFICATION = "多轮意图澄清"
    LONG_TEXT_COMPREHENSION = "长文本语义理解"
    LOGICAL_CONTRADICTION = "逻辑矛盾识别"
    GENERATIVE_GAP_ANALYSIS = "生成式缺失分析"
    ENVIRONMENT_AWARENESS = "环境感知"
    RE_PLANNING = "自主重规划"
    CHAIN_OF_THOUGHT = "链式因果推理"
    GENERATIVE_SOLUTION = "生成式解决方案"


class CognitiveCopilot:
    """
    认知副驾——个人知识学习智能体
    
    这是项目的核心类。它不是一个"带AI功能的笔记软件"，
    而是一个具备自主感知、推理、规划、行动能力的智能体。
    
    五层能力梯度（从被动到主动）：
    L1 反应层：模糊意图挖掘
    L2 关联层：隐性冲突检测  
    L3 生成层：苏格拉底式认知助产
    L4 规划层：动态路径自我修正
    L5 诊断层：归因式学习偏误诊断
    """
    
    def __init__(self, user_id: str):
        self.user_id = user_id
        self.knowledge_base: List[KnowledgeItem] = []
        self.goals: List[LearningGoal] = []
        self.plan: Optional[LearningPlan] = None
        self.diagnosis_history: List[CognitiveDiagnosis] = []
        
        # 智能体状态
        self.is_active = True
        self.last_interaction = datetime.now()
        self.push_preference = "weekly"  # 推送频率：weekly / daily / quiet
        
        print(f"🧠 认知副驾已启动，为用户 {user_id} 服务")
        print(f"📚 已接入 {len(self.knowledge_base)} 条知识条目")
        print(f"⚡ 模型服务：Anthropic Claude（单一模型）")
    
    # ========================================================================
    # L1 反应层：模糊意图挖掘
    # ========================================================================
    
    def fuzzy_recall(self, fragment: str) -> List[KnowledgeItem]:
        """
        功能一：模糊意图的"认知挖掘"
        
        场景：你只记得"橙色封面""好像是讲拖延症的""一个外国老头说的"
        AI动作：不让你精准搜索，而是主动发起多轮追问，逐步澄清意图
        
        AI独特能力：语义推理 + 多轮意图澄清
        信息系统死穴：必须输入精准关键词，否则返回"未找到"
        """
        print(f"\n🔍 [L1-模糊挖掘] 用户说：\"{fragment}\"")
        print("   Agent: 让我帮你回想一下...")
        
        # 实际实现中，这里会调用 Claude API 进行语义匹配和多轮对话
        # 示例返回模拟结果
        results = [
            item for item in self.knowledge_base 
            if fragment.lower() in item.title.lower() 
            or fragment.lower() in item.content[:100].lower()
        ]
        
        if results:
            print(f"   ✅ 找到 {len(results)} 条相关内容")
            for r in results[:2]:
                print(f"      - {r.title} (收藏于 {r.collected_at.strftime('%Y-%m-%d')})")
        else:
            print("   💬 Agent追问：你说的'拖延症'是指心理学话题，还是效率提升？")
        
        return results
    
    # ========================================================================
    # L2 关联层：隐性冲突检测
    # ========================================================================
    
    def detect_conflicts(self) -> List[Dict]:
        """
        功能二：隐性认知"冲突检测"
        
        场景：你半年前收藏了《长期主义者的决策框架》，昨天收藏了《敏捷思维》
        AI动作：全量阅读两篇文章，发现核心观点矛盾，主动推送提醒
        
        AI独特能力：长文本语义理解 + 逻辑矛盾识别
        信息系统死穴：只能按标签分类，无法"读懂"文章内在逻辑
        """
        print("\n⚡ [L2-冲突检测] 正在进行每周扫描...")
        
        # 实际实现：Claude 批量提取主张 → Claude 思考模式做深度逻辑比对
        # 此处模拟检测结果
        conflicts = []
        if len(self.knowledge_base) >= 2:
            conflicts.append({
                "article_a": "长期主义者的决策框架",
                "article_b": "敏捷思维：快速试错",
                "conflict_type": "逻辑矛盾",
                "detail": "A主张'坚守战略至少3年'，B主张'每月根据反馈掉头'",
                "suggestion": "需要我帮你生成一张决策哲学对比表吗？"
            })
            print(f"   ⚠️ 检测到 1 处观点冲突：")
            print(f"      {conflicts[0]['article_a']} ↔ {conflicts[0]['article_b']}")
            print(f"      → {conflicts[0]['detail']}")
        
        return conflicts
    
    # ========================================================================
    # L3 生成层：苏格拉底式认知助产
    # ========================================================================
    
    def generate_deep_questions(self) -> List[str]:
        """
        功能三：苏格拉底式"认知助产"
        
        场景：你知识库里全是"如何开始健身""如何开始写作"——全是入门级
        AI动作：分析缺失项，生成你"该问但没问"的深度问题
        
        AI独特能力：基于知识图谱的缺失项生成高阶问题
        信息系统死穴：能生成统计图表，但无法将"数据缺失"转化为"深刻问题"
        """
        print("\n🤔 [L3-认知助产] 分析你的知识结构...")
        
        # 实际实现：Claude 基于知识图谱生成追问
        questions = []
        
        # 模拟：检测到15篇"如何开始"类文章
        start_articles = [item for item in self.knowledge_base if "开始" in item.title]
        if len(start_articles) > 5:
            questions.append(
                "你在3个不同领域都止步于'入门即放弃'。真正的瓶颈不在于'如何开始'，"
                "而在于'入门后第3-6周的心理机制'。要我把平台期应对的碎片整理成手册吗？"
            )
            print(f"   💡 生成追问：{questions[0][:50]}...")
        
        return questions
    
    # ========================================================================
    # L4 规划层：动态路径自我修正
    # ========================================================================
    
    def check_and_adjust_plan(self) -> Optional[str]:
        """
        功能四：动态"路径自我修正"
        
        场景：AI帮你制定了"三周学Python数据分析"的计划
        AI动作：发现你连续3天没碰Pandas，反而频繁查SQL——主动提议调整
        
        AI独特能力：环境感知 + 自主规划与重规划
        信息系统死穴：项目管理软件让你手动拖拽进度条，无法主动感知和提议
        """
        print("\n🔄 [L4-路径修正] 检查学习计划执行情况...")
        
        if self.plan is None:
            print("   ℹ️ 暂无学习计划，请先设定目标")
            return None
        
        # 模拟：检测到3天未执行 + SQL相关收藏增加
        adjustment = (
            "📢 检测到你连续3天未执行Pandas学习任务，"
            "且近期收藏了3篇SQL优化文章。你手头可能在处理数据库项目？\n"
            "   → 建议临时调整为'SQL进阶周'，Pandas计划顺延。\n"
            "   → 预期学习转化率提升40%。是否同意调整？"
        )
        print(adjustment)
        return adjustment
    
    # ========================================================================
    # L5 诊断层：归因式学习偏误诊断
    # ========================================================================
    
    def diagnose_learning_behavior(self) -> CognitiveDiagnosis:
        """
        功能五：归因式"学习偏误诊断"
        
        场景：你收藏了20篇硬核AI论文，完整读完的只有2篇
        AI动作：不只是展示"完成率10%"，而是链式推理出真实卡点
        
        AI独特能力：多因素归因推理 + 生成式解决方案
        信息系统死穴：只能告诉你"是什么"，永远无法告诉你"为什么"和"怎么办"
        """
        print("\n🏥 [L5-偏误诊断] 正在生成学习健康报告...")
        
        # 实际实现：Claude 思考模式做链式因果推理
        diagnosis = CognitiveDiagnosis(
            pattern="高收藏低完成",
            root_cause="线性代数与概率论前置知识严重缺失——不是懒惰，是跳过了基础",
            confidence=0.92,
            suggested_action="立即暂停收藏新论文，先消化知识库里3篇数学科普文章",
            reasoning_chain=[
                "Step 1: 发现20篇AI论文中仅2篇完整读完（完成率10%）",
                "Step 2: 分析唯二读完的论文，发现用户大量查询数学公式",
                "Step 3: 排除懒惰因素——用户对AI主题有持续的收藏行为",
                "Step 4: 结论——真实卡点是数学基础薄弱"
            ]
        )
        
        print(f"   📊 模式识别：{diagnosis.pattern}")
        print(f"   🎯 归因诊断：{diagnosis.root_cause}")
        print(f"   💊 解决方案：{diagnosis.suggested_action}")
        
        self.diagnosis_history.append(diagnosis)
        return diagnosis
    
    # ========================================================================
    # 智能体主动性——主动推送机制
    # ========================================================================
    
    def generate_weekly_brief(self) -> str:
        """
        主动推送：每周认知简报
        包含：1处冲突检测 + 2个生成式追问
        """
        print("\n📰 [主动推送] 生成本周认知简报...")
        
        conflicts = self.detect_conflicts()
        questions = self.generate_deep_questions()
        
        brief = f"""
╔══════════════════════════════════════════════════════════════════╗
║  📰 认知简报 - 第{datetime.now().isocalendar()[1]}周           ║
╠══════════════════════════════════════════════════════════════════╣
║  ⚡ 冲突检测：{len(conflicts)} 处观点冲突                       ║
║  🤔 追问推荐：{len(questions)} 个值得思考的问题                ║
║  📊 本周活跃度：{len(self.knowledge_base)} 条知识              ║
╚══════════════════════════════════════════════════════════════════╝
        """
        print(brief)
        return brief
    
    # ========================================================================
    # 成本监控
    # ========================================================================
    
    def estimate_monthly_cost(self, user_count: int = 1000) -> Dict:
        """
        估算月成本（基于单一 Claude 服务 + Prompt Caching 与批量合并）
        具体单价在接入 Claude 后回填
        """
        # 占位：以接入后的 Claude 单价估算
        avg_cost_per_user = 0.0  # 元（待实测回填）
        total = avg_cost_per_user * user_count
        
        return {
            "per_user": avg_cost_per_user,
            "total": total,
            "user_count": user_count,
            "cost_controls": {
                "prompt_caching": "系统提示词/知识摘要走缓存",
                "batch_merge": "批量提取/主题分析合并请求",
                "model_tiering": "深度任务与批量任务在供应商内分层"
            }
        }
